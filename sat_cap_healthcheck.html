If your environment has locked /api/ping behind auth # (e.g. via a reverse proxy), set satellite_username/satellite_password # below or via an AAP credential/survey and they'll be used automatically. # # Requires the community.general collection (for the mail module) on the # AAP execution environment: ansible-galaxy collection install community.general - name: Query each Capsule foreman-proxy /version endpoint ansible.builtin.uri: url: "https://{{ item }}:9090/version" method: GET validate_certs: "{{ capsule_validate_certs }}" return_content: true status_code: 200 register: capsule_proxy_versions loop: "{{ capsule_fqdns }}" ignore_errors: true - name: Query each Capsule Pulp status endpoint ansible.builtin.uri: url: "https://{{ item }}/pulp/api/v3/status/" method: GET validate_certs: "{{ capsule_validate_certs }}" return_content: true status_code: 200 register: capsule_pulp_statuses loop: "{{ capsule_fqdns }}" ignore_errors: true - name: Build per-capsule health results ansible.builtin.set_fact: capsule_health_results: "{{ (capsule_health_results | default([])) + [ {'fqdn': pair.0.item, 'foreman_proxy_ok': proxy_ok, 'pulp_workers_ok': workers_ok, 'pulp_content_apps_ok': content_ok, 'pulp_db_ok': db_ok, 'pulp_free_storage_ok': free_ok, 'healthy': (proxy_ok and workers_ok and content_ok and db_ok and free_ok)} ] }}" vars: proxy_ok: "{{ (pair.0.failed | default(false) == false) and (pair.0.json.modules is defined) }}" workers_ok: "{{ (pair.1.failed | default(false) == false) and ((pair.1.json.online_workers | default([]) | length) > 0) }}" content_ok: "{{ (pair.1.failed | default(false) == false) and ((pair.1.json.online_content_apps | default([]) | length) > 0) }}" db_ok: "{{ (pair.1.failed | default(false) == false) and (pair.1.json.database_connection.connected | default(false)) }}" free_ok: "{{ (pair.1.failed | default(false) == false) and ((pair.1.json.storage.free | default(0)) | int >= (pulp_min_free_bytes | int)) }}" # Note: redis_connection intentionally not checked - name: Compute Satellite overall health ansible.builtin.set_fact: satellite_healthy: >- {{ (satellite_db_active | bool == true) and (satellite_katello_status == 'ok') and (satellite_bad_services | length == 0) }} - - name: Email HTML health report community.general.mail: host: "{{ smtp_host }}" port: "{{ smtp_port | default(25) }}" username: "{{ smtp_username | default(omit) }}" password: "{{ smtp_password | default(omit) }}" secure: "{{ smtp_secure | default('starttls') }}" from: "{{ mail_from }}" to: "{{ mail_to }}" subject: >- [{{ 'OK' if overall_healthy else 'ALERT' }}] Satellite/Capsule Health Report - {{ ansible_date_time.date }} subtype: html body: "{{ lookup('file', health_report_path) }}" # ---------------- Fail job if unhealthy (after email is sent) ---------------- - name: Fail if Satellite is unhealthy ansible.builtin.fail: msg: >- Satellite {{ satellite_fqdn }} is UNHEALTHY: db_active={{ satellite_db_active }} katello_status={{ satellite_katello_status }} failing_services={{ satellite_bad_services }} when: not (satellite_healthy | bool) - name: Fail if any Capsule is unhealthy ansible.builtin.fail: msg: >- Unhealthy capsules: {{ capsule_health_results | selectattr('healthy', 'equalto', false) | map(attribute='fqdn') | list }} when: capsule_health_results | selectattr('healthy', 'equalto', false) | list | length > 0
